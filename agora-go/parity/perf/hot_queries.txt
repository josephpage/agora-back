# Hot-path SQL of the Kotlin backend (verbatim, parameters inlined), measured by
# parity/perf/bench_queries.sh before/after the index migration. Format:
# <id>|<description>|<sql on one line>
Q1|top page 1 (/v2/qags top)|SELECT qags.id as id, title, description, post_date as postDate, qags.status, username, thematique_id as thematiqueId, qags.user_id as userId, count(DISTINCT supports_qag.user_id) as supportCount FROM qags LEFT JOIN supports_qag ON qags.id = supports_qag.qag_id WHERE qags.status = 1 GROUP BY qags.id ORDER BY supportCount DESC LIMIT 20 OFFSET 0
Q1t|top page 1, thematique|SELECT qags.id as id, title, description, post_date as postDate, qags.status, username, thematique_id as thematiqueId, qags.user_id as userId, count(DISTINCT supports_qag.user_id) as supportCount FROM qags LEFT JOIN supports_qag ON qags.id = supports_qag.qag_id WHERE qags.status = 1 AND thematique_id = 'th0000000000000000000004' GROUP BY qags.id ORDER BY supportCount DESC LIMIT 20 OFFSET 0
Q2|latest page 1|SELECT qags.id as id, title, description, post_date as postDate, qags.status, username, thematique_id as thematiqueId, qags.user_id as userId, count(DISTINCT supports_qag.user_id) as supportCount FROM qags LEFT JOIN supports_qag ON qags.id = supports_qag.qag_id WHERE qags.status = 1 GROUP BY qags.id ORDER BY post_date DESC LIMIT 20 OFFSET 0
Q3|supporting page 1 (median user)|SELECT qags.id as id, title, description, post_date as postDate, qags.status, username, thematique_id as thematiqueId, qags.user_id as userId, count(DISTINCT supports_qag.user_id) as supportCount FROM qags LEFT JOIN supports_qag ON qags.id = supports_qag.qag_id WHERE ((qags.status = 1 AND qags.id IN (SELECT qag_id FROM supports_qag WHERE user_id = '00000000-0000-4000-9001-00000001f5c6')) OR ((qags.status = 0 OR qags.status = 1) AND qags.user_id = '00000000-0000-4000-9001-00000001f5c6')) GROUP BY qags.id ORDER BY CASE WHEN qags.user_id = '00000000-0000-4000-9001-00000001f5c6' THEN 0 ELSE 1 END, (SELECT support_date FROM supports_qag WHERE qag_id = qags.id AND user_id = '00000000-0000-4000-9001-00000001f5c6' LIMIT 1) DESC LIMIT 20 OFFSET 0
Q4a|count accepted|SELECT count(*) FROM qags WHERE status = 1
Q4b|count accepted, thematique|SELECT count(*) FROM qags WHERE status = 1 AND thematique_id = 'th0000000000000000000004'
Q5|supported count (median user)|SELECT count(*) FROM supports_qag WHERE qag_id IN (SELECT id FROM qags WHERE status = 1 OR (status = 0 AND user_id = '00000000-0000-4000-9001-00000001f5c6')) AND user_id = '00000000-0000-4000-9001-00000001f5c6'
Q6|user supported qags (median user, every list page)|SELECT qag_id FROM supports_qag WHERE qag_id IN (SELECT id FROM qags WHERE status = 1 OR (status = 0 AND user_id = '00000000-0000-4000-9001-00000001f5c6')) AND user_id = '00000000-0000-4000-9001-00000001f5c6' ORDER BY support_date DESC
Q9c|qag details (most supported qag)|SELECT qags.id as id, title, description, post_date as postDate, qags.status, username, thematique_id as thematiqueId, qags.user_id as userId, count(DISTINCT supports_qag.user_id) as supportCount FROM qags LEFT JOIN supports_qag ON qags.id = supports_qag.qag_id WHERE qags.id = '00000000-0000-4000-8001-00000001c5e4' GROUP BY qags.id
Q10|responses list (status 7)|SELECT qags.id as id, title, description, post_date as postDate, qags.status, username, thematique_id as thematiqueId, qags.user_id as userId, count(DISTINCT supports_qag.user_id) as supportCount FROM qags LEFT JOIN supports_qag ON qags.id = supports_qag.qag_id WHERE qags.status = 7 GROUP BY qags.id ORDER BY postDate DESC LIMIT 100
Q12|last user qag (ask_status)|SELECT * FROM qags WHERE (status = 0 OR status = 1) AND user_id = '00000000-0000-4000-9001-000000027435' ORDER BY post_date DESC LIMIT 1
Q13|trending (cache miss)|SELECT qags.id as id, title, description, post_date as postDate, qags.status, username, thematique_id as thematiqueId, qags.user_id as userId, count(DISTINCT supports_qag.user_id) as supportCount, qag_updates.moderated_date as moderatedDate FROM qags LEFT JOIN qag_updates ON qags.id = qag_updates.qag_id LEFT JOIN supports_qag ON qags.id = supports_qag.qag_id WHERE qags.status = 1 AND qag_updates.status = 1 AND qag_updates.moderated_date >= (CURRENT_TIMESTAMP - INTERVAL '7 days') GROUP BY qags.id, qag_updates.moderated_date ORDER BY qag_updates.moderated_date DESC
Q14|is qag supported (support/unsupport)|SELECT * FROM supports_qag WHERE user_id = '00000000-0000-4000-9001-00000001f5c6' AND qag_id = '00000000-0000-4000-8000-000000000001' LIMIT 1
Q17|signup history (suspicious user)|SELECT DISTINCT DATE(event_date) as date, count(*) as signupCount FROM users_data WHERE ip_address_hash = '7d9074fdb306023cccfbb1d43c9ac4a8a8a99954b2381060d5fc8bb1ccd1a07e' AND user_agent = 'Agora/1.4.0 (Android 14; Pixel 8)' AND event_type = 'signup' GROUP BY ip_address_hash, DATE(event_date)
C1|answered consultation ids (/consultations)|SELECT DISTINCT consultation_id FROM user_answered_consultation WHERE user_id = '00000000-0000-4000-9000-000000000024'
C2|has answered consultation (details)|SELECT count(DISTINCT user_id) FROM user_answered_consultation WHERE consultation_id = 'co0000000000000000000001' AND user_id = '00000000-0000-4000-9000-000000000024'
C3|participant count (cache miss)|SELECT count(DISTINCT user_id) FROM user_answered_consultation WHERE consultation_id = 'co0000000000000000000001'
C4|answered count (paginated)|SELECT count(distinct consultation_id) FROM user_answered_consultation WHERE user_id = '00000000-0000-4000-9000-000000000024'
C5|results fallback (cache miss after each submit)|SELECT question_id as questionId, choice_id as choiceId, count(*) as responseCount FROM reponses_consultation WHERE consultation_id = 'co0000000000000000000001' AND choice_id IS NOT NULL GROUP BY (question_id, choice_id)
N1|notifications of a user|select * from notifications where user_id = '00000000-0000-4000-9000-000000000001'
